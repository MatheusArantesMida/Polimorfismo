using System;
using System.Collections.Generic;

namespace Polimorfismo
{
    internal class Program
    {
        static void Main()
        {
            List<INotificacao> notificacoes = new List<INotificacao>
        {
            new NotificacaoEmail(),
            new NotificacaoSMS(),
            new NotificacaoPush()
        };
            foreach (INotificacao notificacao in notificacoes)
            {
                notificacao.Enviar();
            }
            Console.WriteLine("\nDemonstração de sobrecarga:");
            NotificacaoEmail email = new NotificacaoEmail();
            email.Enviar();
            email.Enviar("relatorio.pdf");
        }
    }
}

namespace Polimorfismo
{
    internal interface INotificacao
    {
        void Enviar();
    }
}

using System;

namespace Polimorfismo
{
    internal class NotificacaoEmail : INotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação por email...");
        }
        public void Enviar(string anexo)
        {
            Console.WriteLine($"Enviando email com anexo: {anexo}");
        }
    }
}

using System;

namespace Polimorfismo
{
    internal class NotificacaoSMS : INotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação por SMS...");
        }
    }
}

using System;

namespace Polimorfismo
{
    internal class NotificacaoPush : INotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação push para o app...");
        }
    }
}
